 * Ce script :
 *  - range la liste dans l'onglet "Liste" du Sheet
 *  - garde ton ordre de rayons dans l'onglet "Rayons"

// Le code secret est créé tout seul au premier lancement (onglet "Installation").
// La clé Gemini se met dans Paramètres du projet > Propriétés du script > CLE_GEMINI.
const MODELE_GEMINI = 'gemini-flash-lite-latest';

function proprietes_() { return PropertiesService.getScriptProperties(); }

function codeSecret_() {
  let code = proprietes_().getProperty('CODE_SECRET');
  if (!code) {
    code = Utilities.getUuid().replace(/-/g, '').slice(0, 12);
    proprietes_().setProperty('CODE_SECRET', code);
  }
  return code;
}

function cleGemini_() { return proprietes_().getProperty('CLE_GEMINI') || ''; }

// ---------- Point d'entrée ----------

function doGet(e) {
  return repondre_({ ok: true, message: 'Le moteur Courses fonctionne.' });
}

function doPost(e) {
  let req;
  try {
    req = JSON.parse(e.postData.contents);
  } catch (err) {
    return repondre_({ ok: false, erreur: 'Requête illisible.' });
  }
  if (!req || req.secret !== codeSecret_()) {
    return repondre_({ ok: false, erreur: 'Code secret incorrect.' });
  }

  try {
    switch (req.action) {
      case 'liste':
        return repondre_(etat_());
      case 'ajouter':
        return repondre_(ajouter_(req));
      case 'retirer':
        return repondre_(avecVerrou_(() => { retirer_(req.id); return etat_(); }));
      case 'modifier':
        return repondre_(avecVerrou_(() => { modifier_(req); return etat_(); }));
      case 'rayons':
        return repondre_(avecVerrou_(() => { enregistrerRayons_(req.rayons); return etat_(); }));
      case 'analyser':
        return repondre_(analyser_(req.texte));
      default:
        return repondre_({ ok: false, erreur: 'Action inconnue.' });
    }
  } catch (err) {
    return repondre_({ ok: false, erreur: String(err && err.message || err) });
  }
}

function repondre_(obj) {
  if (obj.ok === undefined) obj.ok = true;
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function avecVerrou_(fn) {
  const verrou = LockService.getScriptLock();
  verrou.waitLock(15000);
  try { return fn(); } finally { verrou.releaseLock(); }
}

// ---------- Onglets du Sheet ----------

function onglet_(nom, entetes) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(nom);
  if (!sh) {
    sh = ss.insertSheet(nom);
    sh.getRange(1, 1, 1, entetes.length).setValues([entetes]).setFontWeight('bold');
    sh.setFrozenRows(1);
  } else if (sh.getLastColumn() < entetes.length) {
    // ajoute les colonnes apparues dans une nouvelle version
    sh.getRange(1, 1, 1, entetes.length).setValues([entetes]).setFontWeight('bold');
  }
  return sh;
}

const COLONNES_LISTE = ['id', 'article', 'quantité', 'rayon', 'ajouté par', 'date', 'sur facture'];
function ongletListe_() { return onglet_('Liste', COLONNES_LISTE); }
function ongletMemoire_() { return onglet_('Memoire', ['article', 'rayon', 'sur facture']); }
function ongletRayons_() {
  const sh = onglet_('Rayons', ['Ordre de mes rayons (modifiable)']);
  if (sh.getLastRow() < 2) {
    sh.getRange(2, 1, RAYONS_DEFAUT.length, 1).setValues(RAYONS_DEFAUT.map(r => [r]));
  }
  return sh;
}

function lignes_(sh, nbColonnes) {
  const n = sh.getLastRow() - 1;
  if (n < 1) return [];
  return sh.getRange(2, 1, n, nbColonnes || sh.getLastColumn()).getValues();
}

function rayons_() {
  const liste = lignes_(ongletRayons_(), 1).map(l => String(l[0]).trim()).filter(Boolean);
  if (liste.indexOf('Autre') === -1) liste.push('Autre');
  return liste;
}

const vrai_ = v => v === true || String(v).toLowerCase() === 'oui' || String(v).toLowerCase() === 'true';

function etat_() {
  const articles = lignes_(ongletListe_(), COLONNES_LISTE.length).filter(l => l[0]).map(l => ({
    id: String(l[0]), nom: String(l[1]), quantite: String(l[2] || ''),
    rayon: String(l[3] || 'Autre'), par: String(l[4] || ''),
    date: l[5] instanceof Date ? l[5].getTime() : 0,
    facture: vrai_(l[6])
  }));
  return { ok: true, articles: articles, rayons: rayons_() };
}

// ---------- Ajouter ----------

/**
 * req.texte : "lait, 2 baguettes et du beurre" (app ou Siri)
 * req.articles : [{nom, quantite, rayon}] (après l'analyse IA)
 */
function ajouter_(req) {
  return avecVerrou_(() => {
    const sh = ongletListe_();
    const rayons = rayons_();
    const memoire = memoire_();
    const existants = {};
    lignes_(sh, 2).forEach(l => { existants[normaliser_(l[1])] = true; });

    let aAjouter = [];
    if (Array.isArray(req.articles)) {
      aAjouter = req.articles.map(a => ({
        nom: propre_(a.nom), quantite: String(a.quantite || '').trim(), rayon: a.rayon
      }));
    } else {
      aAjouter = decouper_(String(req.texte || '')).map(lireQuantite_);
    }

    const ajoutes = [];
    const nouvelles = [];
    aAjouter.forEach(a => {
      if (!a.nom) return;
      const cle = normaliser_(a.nom);
      if (existants[cle]) return; // déjà dans la liste
      existants[cle] = true;
      const souvenir = memoire[cle] || {};
      const connu = r => r && rayons.indexOf(r) !== -1 ? r : '';
      let rayon = connu(souvenir.rayon) || connu(a.rayon) || deviner_(cle);
      if (rayons.indexOf(rayon) === -1) rayon = 'Autre';
      const facture = souvenir.facture !== undefined ? souvenir.facture : RAYONS_FACTURE.indexOf(rayon) !== -1;
      const nom = a.nom.charAt(0).toUpperCase() + a.nom.slice(1);
      nouvelles.push([Utilities.getUuid().slice(0, 8), nom, a.quantite, rayon, String(req.par || ''), new Date(), facture ? 'oui' : '']);
      ajoutes.push(nom);
    });
    if (nouvelles.length) {
      sh.getRange(sh.getLastRow() + 1, 1, nouvelles.length, COLONNES_LISTE.length).setValues(nouvelles);
    }

    const etat = etat_();
    etat.message = ajoutes.length
      ? ajoutes.join(', ') + (ajoutes.length > 1 ? ' ajoutés' : ' ajouté') + ' à la liste.'
      : 'C\'était déjà sur la liste.';
    return etat;
  });
}

function decouper_(texte) {
  return texte
    .split(/\n|,|;|\bpuis\b|\bet\b(?! (?:demi|quart))/i)
    .map(s => s.replace(/^\s*[-•*]\s*/, '').trim())
    .filter(Boolean);
}

function lireQuantite_(morceau) {
  let s = morceau.trim();
  s = s.replace(/^(ajoute|rajoute|mets?|achete|acheter)\s+/i, '');
  let quantite = '';
  const m = s.match(/^(\d+(?:[.,]\d+)?)\s*(kg|g|l|cl|ml|x)?\s+(.*)$/i);
  if (m) {
    quantite = m[1] + (m[2] ? ' ' + m[2].toLowerCase() : '');
    s = m[3];
  }
  s = s.replace(/^(?:(?:de l|d|l)['’]\s*|(?:du|des|de la|de|un|une|le|la|les)\s+)/i, '');
  return { nom: propre_(s), quantite: quantite };
}

function propre_(s) {
  return String(s || '').replace(/\s+/g, ' ').replace(/[.!?]+$/, '').trim().toLowerCase();
}

// ---------- Deviner le rayon ----------

function normaliser_(s) {
  return String(s || '').toLowerCase().replace(/œ/g, 'oe').replace(/æ/g, 'ae')
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .replace(/[’']/g, ' ').replace(/[^a-z0-9 -]/g, ' ')
    .split(/\s+/).filter(Boolean)
    .map(m => m.length > 3 ? m.replace(/(s|x)$/, '') : m)
    .join(' ');
}

let dico_ = null;
function deviner_(cle) {
  if (!dico_) {
    dico_ = [];
    Object.keys(MOTS_RAYONS).forEach(rayon => {
      MOTS_RAYONS[rayon].split(',').forEach(mot => {
        const c = normaliser_(mot);
        if (c) dico_.push({ cle: c, rayon: rayon });
      });
    });
    // l'expression la plus longue gagne : "lait d'amande" passe avant "lait"
    dico_.sort((a, b) => b.cle.length - a.cle.length);
  }
  const entoure = ' ' + cle + ' ';
  for (let i = 0; i < dico_.length; i++) {
    if (entoure.indexOf(' ' + dico_[i].cle + ' ') !== -1) return dico_[i].rayon;
  }
  return 'Autre';
}

// { "lait": { rayon: "Lait & œufs", facture: false }, ... }
function memoire_() {
  const m = {};
  lignes_(ongletMemoire_(), 3).forEach(l => {
    if (!l[0]) return;
    m[String(l[0])] = {
      rayon: String(l[1] || '') || undefined,
      facture: l[2] === '' ? undefined : vrai_(l[2])
    };
  });
  return m;
}

function retenir_(nom, changements) {
  const sh = ongletMemoire_();
  const cle = normaliser_(nom);
  const lignes = lignes_(sh, 3);
  let index = -1;
  for (let i = 0; i < lignes.length; i++) {
    if (String(lignes[i][0]) === cle) { index = i + 2; break; }
  }
  if (index === -1) {
    sh.appendRow([cle, '', '']);
    index = sh.getLastRow();
  }
  if (changements.rayon) sh.getRange(index, 2).setValue(changements.rayon);
  if (typeof changements.facture === 'boolean') sh.getRange(index, 3).setValue(changements.facture ? 'oui' : 'non');
}

// ---------- Retirer / modifier ----------

function trouverLigne_(sh, id) {
  const lignes = lignes_(sh, COLONNES_LISTE.length);
  for (let i = 0; i < lignes.length; i++) {
    if (String(lignes[i][0]) === String(id)) return { index: i + 2, ligne: lignes[i] };
  }
  return null;
}

function retirer_(id) {
  const sh = ongletListe_();
  const t = trouverLigne_(sh, id);
  if (t) sh.deleteRow(t.index);
}

function modifier_(req) {
  const sh = ongletListe_();
  const t = trouverLigne_(sh, req.id);
  if (!t) return;
  const nom = (typeof req.nom === 'string' && req.nom.trim()) ? req.nom.trim() : String(t.ligne[1]);
  if (typeof req.quantite === 'string') sh.getRange(t.index, 3).setValue(req.quantite.trim());
  if (nom !== t.ligne[1]) sh.getRange(t.index, 2).setValue(nom);
  const souvenir = {};
  if (req.rayon && rayons_().indexOf(req.rayon) !== -1) {
    sh.getRange(t.index, 4).setValue(req.rayon);
    souvenir.rayon = req.rayon;
  }
  if (typeof req.facture === 'boolean') {
    sh.getRange(t.index, 7).setValue(req.facture ? 'oui' : '');
    souvenir.facture = req.facture;
  }
  // la prochaine fois, ce sera direct le bon rayon / sur facture
  if (souvenir.rayon || souvenir.facture !== undefined) retenir_(nom, souvenir);
}

function enregistrerRayons_(liste) {
  if (!Array.isArray(liste)) return;
  const propres = liste.map(r => String(r).trim()).filter(Boolean);
  if (!propres.length) return;
  const sh = ongletRayons_();
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 1).clearContent();
  sh.getRange(2, 1, propres.length, 1).setValues(propres.map(r => [r]));
}

// ---------- IA : lire un message collé ----------

function analyser_(texte) {
  texte = String(texte || '').slice(0, 4000);
  if (!texte.trim()) return { ok: false, erreur: 'Le message est vide.' };
  const rayons = rayons_();

  const cle = cleGemini_();
  if (!cle) {
    // Sans clé : découpage simple, sans correction des fautes
    return {
      ok: true, ia: false,
      articles: decouper_(texte).map(lireQuantite_).filter(a => a.nom)
        .map(a => ({ nom: a.nom, quantite: a.quantite, rayon: deviner_(normaliser_(a.nom)) }))
    };
  }

  const consigne =
    'Tu reçois un message (SMS, WhatsApp...) qui contient une liste de courses, ' +
    'souvent avec des fautes d\'orthographe, des abréviations et du blabla. ' +
    'Extrais UNIQUEMENT les articles à acheter. Corrige l\'orthographe, écris chaque article ' +
    'au plus simple en français (ex : "pq" -> "papier toilette", "yaourt natur" -> "yaourts nature"), ' +
    'garde la quantité si elle est donnée (ex : "2", "1 kg", "6"), sinon chaîne vide. ' +
    'Choisis pour chaque article un rayon EXACTEMENT parmi cette liste : ' + JSON.stringify(rayons) + '. ' +
    'Réponds seulement avec un tableau JSON de la forme ' +
    '[{"nom":"lait","quantite":"2","rayon":"Lait & œufs"}].\n\nMessage :\n' + texte;

  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' +
    encodeURIComponent(MODELE_GEMINI) + ':generateContent?key=' + encodeURIComponent(cle);
  const rep = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    muteHttpExceptions: true,
    payload: JSON.stringify({
      contents: [{ parts: [{ text: consigne }] }],
      generationConfig: { temperature: 0, responseMimeType: 'application/json' }
    })
  });

  if (rep.getResponseCode() !== 200) {
    return { ok: false, erreur: 'L\'IA n\'a pas répondu (' + rep.getResponseCode() + '). Réessaie dans un moment.' };
  }
  const corps = JSON.parse(rep.getContentText());
  const brut = corps.candidates && corps.candidates[0] && corps.candidates[0].content.parts[0].text || '[]';
  let articles = [];
  try { articles = JSON.parse(brut); } catch (err) { articles = []; }
  if (!Array.isArray(articles)) articles = [];

  const memoire = memoire_();
  articles = articles.filter(a => a && a.nom).map(a => {
    const nom = propre_(a.nom);
    const cle = normaliser_(nom);
    const souvenir = memoire[cle] || {};
    const rayon = souvenir.rayon || (rayons.indexOf(a.rayon) !== -1 ? a.rayon : deviner_(cle));
    return { nom: nom, quantite: String(a.quantite || '').trim(), rayon: rayon };
  });
  return { ok: true, ia: true, articles: articles };
}

// À lancer une fois depuis l'éditeur : crée les onglets et le code secret.
function installer() {
  ongletListe_(); ongletRayons_(); ongletMemoire_();
  const sh = onglet_('Installation', ['À recopier dans l\'app sur chaque iPhone']);
  sh.getRange(2, 1, 3, 2).setValues([
    ['Code secret', codeSecret_()],
    ['Adresse du script', 'Déployer > Gérer les déploiements > URL (finit par /exec)'],
    ['Clé IA', cleGemini_() ? 'installée' : 'pas encore (Paramètres du projet > Propriétés du script > CLE_GEMINI)']
  ]);
  sh.autoResizeColumns(1, 2);
  const vide = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Feuille 1') || SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Sheet1');
  if (vide && vide.getLastRow() === 0) SpreadsheetApp.getActiveSpreadsheet().deleteSheet(vide);
}

// À lancer depuis l'éditeur pour remettre le parcours du Leclerc de Limoux
// et reclasser les articles déjà dans la liste.
function appliquerParcoursLimoux() {
  installer();
  enregistrerRayons_(RAYONS_DEFAUT);
  const sh = ongletListe_();
  const memoire = memoire_();
  lignes_(sh, COLONNES_LISTE.length).forEach((l, i) => {
    if (!l[0]) return;
    const cle = normaliser_(l[1]);
    const souvenir = memoire[cle] || {};
    const rayon = (souvenir.rayon && RAYONS_DEFAUT.indexOf(souvenir.rayon) !== -1) ? souvenir.rayon : deviner_(cle);
    const facture = souvenir.facture !== undefined ? souvenir.facture : (vrai_(l[6]) || RAYONS_FACTURE.indexOf(rayon) !== -1);
    sh.getRange(i + 2, 4).setValue(rayon);
    sh.getRange(i + 2, 7).setValue(facture ? 'oui' : '');
  });
}
